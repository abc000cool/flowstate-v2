# Table 4 stations
def station_table(seeds,label):
    sim=np.array([rows[s][2]["segment_speeds_sim"] for s in seeds],float)  # seeds x 42 x 14
    ob=np.array([rows[s][2]["segment_speeds_obs"] for s in seeds],float)
    geh=np.array([rows[s][2]["geh_values"] for s in seeds],float).reshape(len(seeds),14,3)
    err=((sim-ob)/ob)**2
    tot=np.nansum(err)
    print(f"\n== T4 stations, {label}: id | label | x | sim mean | obs mean | RMSPE_station | share of SSE | GEH mean (h1,h2,h3) | first window sim<10 m/s | first window obs<10 m/s | windows sim<10 | obs<10")
    for i,st in enumerate(stations):
        e=err[:,:,i]; ss=np.nanmean(sim[:,:,i]); oo=np.nanmean(ob[:,:,i])
        rm=math.sqrt(np.nanmean(e)); share=np.nansum(e)/tot
        simw=np.nanmean(sim[:,:,i],axis=0); obw=np.nanmean(ob[:,:,i],axis=0)
        wins=rows[seeds[0]][2]["windows"]
        def first(v):
            idx=[wins[k] for k in range(len(v)) if v[k]<10]
            return f"w{idx[0]} ({5.5+idx[0]/12:.2f}h)" if idx else "-"
        print(f"{st['id']} | {st['label']} | {st['x_m']:.0f} | {ss:.1f} | {oo:.1f} | {rm:.3f} | {share:.3f} | {geh[:,i,:].mean(axis=0).round(1).tolist()} | {first(simw)} | {first(obw)} | {(simw<10).sum()} | {(obw<10).sum()}")
    print("overall rmspe", math.sqrt(np.nanmean(err)))
    return sim,ob
