P18_ARM=scenarios/i24_replica_flow_rc_speedcal_dc_refit.yaml

P18_REF=artifacts/i24_validation_dc_refit_rc.json
P18_S2=artifacts/driver_joint_s2_i24.json
P18_OUT=artifacts/driver_joint_i24.json
P18_TARGET=1775
P18_RUN=scripts/driver_joint_run.py   # NOT BUILT (see the header)
p18_guard() {  # the plan's Stop rule, then the populations' lineage
  $RUN scripts/driver_joint_screen.py --check-stop
  local rc=$?
  if [ "$rc" -eq 10 ]; then say "p18: S1 admits no pair with k > 0 - the plan's Stop rule; nothing run"; return 1; fi
  [ "$rc" -eq 0 ] || { say "p18: the S1 artifact could not be read (exit $rc); nothing run"; return 1; }
  $RUN scripts/driver_joint_grid.py --check || { say "p18: a population is not the recorded derivation; nothing run"; return 1; }
}
p18_s2_one() {  # p18_s2_one <k> <j> <population>: the capacity fixture on one pair, two seeds (resumable)
  local d="runs/p18/s2/k$1_j$2"
  [ -f "$d/population.calibration.json" ] && return 0
  mkdir -p "$d"
  $RUN scripts/calibrate_capacity.py --source "$3" --out "$d/population.json" --target "$P18_TARGET" --lanes 4 \
      --base-scenario "$P18_ARM" --t-scales 1.0 --seeds 1 2 --procs 2 > "$d/calibrate_capacity.log" 2>&1
}
p18_s2() {
  local k j pop par=$((PROCS / 2))
  [ "$par" -ge 1 ] || par=1
  mkdir -p runs/p18/s2
  $RUN scripts/driver_joint_screen.py --list-pass > runs/p18/s1_pass.txt || return 1
  while read -r k j pop; do
    p18_s2_one "$k" "$j" "$pop" &
    while [ "$(jobs -rp | wc -l)" -ge "$par" ]; do wait -n; done
  done < runs/p18/s1_pass.txt
  wait
  $RUN scripts/driver_joint_s2.py --runs runs/p18/s2 --screen artifacts/driver_joint_screen_i24.json \
      --target "$P18_TARGET" --out "$P18_S2" || { say "p18: S2 reading failed"; return 1; }   # NOT BUILT
}
p18_stage3() {  # p18_stage3 <rank>: the rank-th choice's base copy, B5 refit, 20-seed battery, readout
  local rank="$1" k j pop tag base scn fit label
  read -r k j pop tag < <($RUN "$P18_RUN" select --artifact "$P18_OUT" --rank "$rank") \
    || { say "p18: no pair of rank $rank to take to stage 3"; return 1; }
  base="scenarios/i24_replica_flow_rc_corrected_b6_$tag.yaml"
  scn="scenarios/i24_replica_flow_rc_speedcal_b6_${tag}_refit.yaml"
  fit="artifacts/demand_level_i24_b6_$tag.json"
  label="p18_b6_$tag"
  $RUN "$P18_RUN" base-copy --source "$P18_BASE" --source-hash "$P18_BASE_HASH" --population "$pop" --out "$base" \
    || { say "p18: no stage-3 base for $tag"; return 1; }
  if [ -f "logs/p18_fit_$tag.ok" ] && [ -f "$fit" ] && [ -f "$scn" ]; then
    say "p18: the fit on $tag was done earlier ($fit), not repeated"
  else
    $RUN scripts/fit_demand_level.py --corridor i24 --base-yaml "$base" --fleet-artifact "$pop" --from-battery "$P18_REF" \
        --procs "$PROCS" --out "$fit" --write-scenario --scenario-out "$scn" --name "$(basename "$scn" .yaml)" \
      || { say "p18: B5's fit on $tag failed or flagged constraint_unmet; its battery skipped"; return 1; }   # ASSUMED (B5)
    touch "logs/p18_fit_$tag.ok"
  fi
  if ! { [ -f "logs/$label.battery.ok" ] && [ -f "artifacts/i24_validation_$label.json" ]; }; then
    $RUN scripts/i24_validate.py --scenario "$scn" --label "$label" --replicates "$REPS" --procs "$PROCS" \
        --analysis-procs 8 --ring-seeds "$RING" || { say "p18: battery $label failed"; return 1; }
    p4_prune "runs/i24_validation/$label"
    touch "logs/$label.battery.ok"
  fi
  $RUN "$P18_RUN" stage3 --battery "artifacts/i24_validation_$label.json" --reference "$P18_REF" \
      --reference-hash "$P18_ARM_HASH" --fit "$fit" --out "$P18_OUT" || { say "p18: stage-3 readout of $tag failed"; return 1; }
}
p18_steps() {
  p18_guard || return 1
  p18_s2 || return 1
  $RUN "$P18_RUN" run --arm "$P18_ARM" --arm-hash "$P18_ARM_HASH" --pairs-from "$P18_S2" --reference-pair 1.0 0 \
      --seeds 3 --procs "$PROCS" --mem-per-run-gb 4.5 --out runs/p18/grid || { say "p18: grid run failed"; return 1; }
  $RUN "$P18_RUN" select --artifact "$P18_OUT" --runs runs/p18/grid --write || { say "p18: no pair selected"; return 1; }
  make_archive light   # the grid and its readout leave the machine before stage 3
  p18_stage3 1
}
p18b_steps() {  # the second pair, in its own launch (plan B6 step 5: at most two pairs, the second separately)
  p18_guard || return 1
  p18_stage3 2
}
if echo " $STAGES " | grep -q " p18_i24_b6 "; then
  stage p18_i24_b6 p18_steps || say "p18_i24_b6 failed; continuing"
fi
if echo " $STAGES " | grep -q " p18b_i24_b6 "; then
  stage p18b_i24_b6 p18b_steps || say "p18b_i24_b6 failed; continuing"
fi
