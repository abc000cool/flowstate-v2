P25_H=artifacts/i24_discharge_2026-10-07/harness_b2_k0/corridor_b2_k0.py
P25_B2H=artifacts/i24_discharge_2026-10-07/harness_b2/corridor_b2.py
P25_BRAKE=artifacts/i24_discharge_2026-10-07/harness/hard_braking.py
P25_REF_SCN=scenarios/i24_replica_flow_speedcal.yaml
P25_ARM=scenarios/i24_replica_flow_rc_speedcal.yaml
P25_REF=flow_speedcal_p25ref
P25_RC=flow_speedcal_rc
p25_battery() {  # p25_battery <scenario> <label>: stage p4_i24_ref's battery line, its braking counts and ramp flows, once
  local ok="logs/p25_$2.battery.ok"
  if [ -f "$ok" ] && [ -f "artifacts/i24_validation_$2.json" ] && [ -f "artifacts/i24_b2_ramp_flows_$2.json" ] \
      && [ -f "artifacts/i24_hard_braking_$2.json" ]; then
    say "p25: battery $2 done earlier, not repeated"; return 0
  fi
  $RUN scripts/i24_validate.py --scenario "$1" --label "$2" --replicates "$REPS" --procs "$PROCS" \
      --analysis-procs 8 --ring-seeds "$RING" || { say "p25: battery $2 failed"; return 1; }
  $RUN "$P25_BRAKE" --artifact "artifacts/i24_validation_$2.json" --out "artifacts/i24_hard_braking_$2.json" \
    || { say "p25: braking counts of $2 failed"; return 1; }
  $RUN "$P25_B2H" reduce --battery "artifacts/i24_validation_$2.json" --out "artifacts/i24_b2_ramp_flows_$2.json" \
    || { say "p25: ramp flows of $2 failed"; return 1; }
  p4_prune "runs/i24_validation/$2"
  touch "$ok"
}
p25_steps() {
  $RUN "$P25_H" arm --out "$P25_ARM" \
    || { say "p25: the k = 0 B2 arm was refused (reason in logs/p25_i24_b2_k0.log); nothing run"; return 1; }
  p25_battery "$P25_REF_SCN" "$P25_REF" || return 1
  $RUN "$P25_H" check-ref --battery "artifacts/i24_validation_$P25_REF.json" \
    || { say "p25: the k = 0 re-run does not reproduce artifacts/i24_validation_flow_speedcal_ref.json; the arm is not run; the owner decides"; return 1; }
  make_archive light   # the reference leaves the machine before the arm starts
  p25_battery "$P25_ARM" "$P25_RC" || return 1
  make_archive light
  $RUN "$P25_H" score --out artifacts/boundary_b2_k0_corridor.json \
    || { say "p25: the readout is blocked or failed (artifacts/boundary_b2_k0_corridor.json says why)"; return 1; }
}
if echo " $STAGES " | grep -q " p25_i24_b2_k0 "; then
  stage p25_i24_b2_k0 p25_steps || say "p25_i24_b2_k0 failed; continuing"
fi
