P22_H=scripts/regenerate_report.py
P22_SEED=6914975401685141156
P22_OUT=runs/p22
P22_ARMS="${P22_ARMS_ARG:-e13_i24_b2:scenarios/i24_replica_flow_rc_speedcal_dc_refit.yaml:artifacts/i24_validation_dc_refit_rc_p23.json \
e13_i24_b5::artifacts/i24_validation_p15_b5.json \
e13_i94_rbc:scenarios/mndot_i94_wb_stpaul_weave_dc_cal_w1b_w2_rbc.yaml:artifacts/validation_mndot_i94_wb_stpaul_weave_xlsfg_dc_cal_w1b_w2_rbc_gated.json \
e13_i94_b5::artifacts/validation_mndot_i94_wb_stpaul_weave_xlsfg_dc_cal_w1b_w2_rbc_b5_gated.json \
e13_ring:ring_sugiyama:artifacts/i24_validation_dc_refit_rc.json}"
p22_steps() {
  local rc=0 spec line
  local -a arms=()
  for spec in $P22_ARMS; do arms+=(--arm "$spec"); done
  mkdir -p "$P22_OUT"
  # ${arms[@]+...}: an empty array under set -u (bash 3.2 calls it unbound)
  $RUN "$P22_H" plan --seed "$P22_SEED" --out "$P22_OUT" ${arms[@]+"${arms[@]}"} > "$P22_OUT/plan.log" 2>&1 || rc=1
  while IFS= read -r line; do
    case "$line" in "p22: "*) say "$line" ;; *) echo "$line" ;; esac
  done < "$P22_OUT/plan.log"
  [ -f "$P22_OUT/PLAN.json" ] || { say "p22: no plan was written (logs/p22_reports.log says why); nothing run"; return 1; }
  if [ -s "$P22_OUT/pairs.txt" ]; then
    # shellcheck disable=SC2046  # one argument per line, none with a space
    $RUN scripts/run_pairs.py --out "$P22_OUT" --procs "$PROCS" $(cat "$P22_OUT/pairs.txt") \
      || { say "p22: a figure replicate failed or was refused (runs/p22/PAIRS.json says which)"; rc=1; }
  fi
  $RUN "$P22_H" ring-run --plan "$P22_OUT/PLAN.json" || { say "p22: the ring replicate failed"; rc=1; }
  $RUN "$P22_H" render --plan "$P22_OUT/PLAN.json" --record artifacts/p22_reports.json --i24-observed \
    || { say "p22: a planned report was not written (artifacts/p22_reports.json says which)"; rc=1; }
  return $rc
}
if echo " $STAGES " | grep -q " p22_reports "; then
  stage p22_reports p22_steps || say "p22_reports failed; continuing"
fi
